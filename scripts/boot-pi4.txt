setenv autostart yes

# The Raspberry Pi 4's physical SD card slot is driven by the "emmc2"
# controller, while mmc@7e300000 is actually the onboard WiFi SDIO chip (see
# bcm2711-rpi-4-b.dts's &sdhci node, "SDHCI is used to control the SDIO for
# wireless"). That WiFi device is non-removable and always logs a benign
# "Card did not respond to voltage select!" during bootflow scanning. Which
# u-boot mmc index (0 or 1) ends up assigned to emmc2 depends on the node
# order of the devicetree that was loaded (it flips between u-boot's own
# bundled bcm2711-rpi-4-b.dtb and Raspberry Pi's official firmware one), so
# try both instead of hardcoding a single index.
#
# This script itself can be booted from either the microSD card (mmc) or a
# USB mass-storage device (e.g. behind the VL805 xHCI controller): the same
# disk.img layout is written to either medium by tools-pi4/create_sdcard.sh.
# When booting from USB, u-boot's "bootflow"/EFI boot-manager scan (visible
# earlier in the boot log as "starting USB... USB XHCI ... scanning usb for
# storage devices...") has already enumerated the USB device before this
# script runs, so no explicit "usb start" should be needed here; call it
# anyway (ignoring failure) in case this script is ever invoked from a
# context where that scan did not happen, so `load usb ...` below still has
# a populated device list to read from.
usb start
if load mmc 0:1 $kernel_addr_r mini.elf || load mmc 1:1 $kernel_addr_r mini.elf || load usb 0:1 $kernel_addr_r mini.elf; then
	bootelf $kernel_addr_r $fdt_addr $kernel_addr_r
else
	echo "Unable to read mini.elf"
fi
sleep 10000000
