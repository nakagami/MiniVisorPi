setenv autostart yes

# This script itself can be booted from either the microSD card (mmc) or a
# USB mass-storage device (e.g. behind the VL805 xHCI controller): the same
# disk.img layout is written to either medium by tools-pi4/create_sdcard.sh.
#
# Before running a "script" bootflow, u-boot's bootstd layer already knows
# exactly which device/partition it read this file from and exports that as
# ${devtype}/${devnum}/${distro_bootpart} (see
# boot/bootmeth_script.c:script_boot()). Print them so this is visible on
# real hardware, and prefer them over guessing/probing "mmc 0:1", "mmc 1:1",
# "usb 0:1" in turn when they are available:
#  - The Raspberry Pi 4's physical SD card slot is driven by the "emmc2"
#    controller, while mmc@7e300000 is actually the onboard, non-removable
#    WiFi SDIO chip (see bcm2711-rpi-4-b.dts's &sdhci node); probing it always
#    logs a benign but slow "Card did not respond to voltage select!" /
#    "mmc1 busy timeout" retry sequence (multiple seconds) during bootflow
#    scanning, and which mmc index (0 or 1) is the real SD slot flips
#    depending on devicetree node order.
#  - On real hardware, a failed "load mmc ..." attempt right before "load usb
#    0:1 ..." appears to leave the USB mass-storage device unable to be read
#    afterwards, even though the exact same "load usb 0:1 ..." succeeds when
#    typed by hand right after "usb start" with no preceding mmc attempt. So
#    when ${devtype}/${devnum} are not available and we have to probe, try
#    USB *before* any mmc device to avoid that ordering problem.
#
# Each attempt below uses its own single-condition "if load ...; then
# bootelf ...; fi" (bootelf never returns on success) instead of chaining
# multiple "load" attempts together with "||"/"&&" in one condition, to avoid
# any ambiguity in how this build's hush parser scores a multi-way
# and/or list for the purposes of ${?}/if_code.
echo "boot-pi4: devtype=$devtype devnum=$devnum distro_bootpart=$distro_bootpart"

if test -n "$devtype"; then
	if load ${devtype} ${devnum}:${distro_bootpart} $kernel_addr_r mini.elf; then
		bootelf $kernel_addr_r $fdt_addr $kernel_addr_r
	fi
	echo "Unable to read mini.elf from ${devtype} ${devnum}:${distro_bootpart}"
else
	usb start
	if load usb 0:1 $kernel_addr_r mini.elf; then
		bootelf $kernel_addr_r $fdt_addr $kernel_addr_r
	fi
	if load mmc 0:1 $kernel_addr_r mini.elf; then
		bootelf $kernel_addr_r $fdt_addr $kernel_addr_r
	fi
	if load mmc 1:1 $kernel_addr_r mini.elf; then
		bootelf $kernel_addr_r $fdt_addr $kernel_addr_r
	fi
	echo "Unable to read mini.elf"
fi
sleep 10000000
